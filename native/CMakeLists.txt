cmake_minimum_required(VERSION 3.16)
project(reversi_native LANGUAGES CXX)
set(NATIVE_MODEL_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.local-research/native" CACHE PATH "Directory containing the verified generated frozen.hpp")
if(NOT EXISTS "${NATIVE_MODEL_DIR}/frozen.hpp")
  message(FATAL_ERROR "Generate verified models first: node scripts/native/build.mjs")
endif()
add_executable(reversi-native engine.cpp main.cpp)
target_compile_features(reversi-native PRIVATE cxx_std_17)
target_include_directories(reversi-native PRIVATE "${NATIVE_MODEL_DIR}")
if(MSVC)
  message(FATAL_ERROR "Use Clang or GCC (WSL2 is the primary Windows environment). MSVC is not qualified.")
else()
  target_compile_options(reversi-native PRIVATE -O3 -Wall -Wextra -Wpedantic -ffp-contract=off -fno-fast-math)
endif()
